%% Entry macro: \cventry{dates}{title}{org}{location}{description}
\newcommand{\cventry}[5]{%
  \def\cventryDesc{#5}%
  \begin{tabularx}{\linewidth}{@{}p{2.5cm} X@{}}
    {\small\color{hkugray}#1} &
    \begin{minipage}[t]{\linewidth}
      {\textbf{#2}} \hfill {\small\color{hkugray}\textit{#4}}\\
      {\color{hkugreen}#3}%
      \ifx\cventryDesc\empty\else
        \vspace{2pt}\\{\small\color{hkugray}#5}%
      \fi
    \end{minipage}
  \end{tabularx}\vspace{4pt}}

%% Publication entry: \pubentry{authors}{title}{venue}{year}{url}
\newcommand{\pubentry}[5]{%
  \def\pubentryURL{#5}%
  \begin{tabularx}{\linewidth}{@{}p{2.5cm} X@{}}
    {\small\color{hkugray}#4} &
    {\small #1, ``\textit{#2}'', \textbf{#3}.%
    \ifx\pubentryURL\empty\else~\href{#5}{\faLink}\fi}
  \end{tabularx}\vspace{3pt}}
